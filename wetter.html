async function fetchPWSData() { const apiKey = 'd3face96b217434bbace96b217734b30'; const stationId = 'IMAINB38'; const targetUrl = `https://api.weather.com/v2/pws/observations/current?stationId=${stationId}&format=json&units=m&apiKey=${apiKey}`; // Stabiler CORS-Proxy const proxyUrl = `https://corsproxy.io/?${encodeURIComponent(targetUrl)}`; const statusMsg = document.getElementById('pwsStatusMsg'); const statusDot = document.getElementById('statusDot'); try { statusMsg.innerText = 'Rufe Live-Daten ab...'; const response = await fetch(proxyUrl); if (!response.ok) throw new Error(`HTTP Status ${response.status}`); const data = await response.json(); if (data && data.observations && data.observations.length > 0) { const obsGeneral = data.observations[0]; const obs = obsGeneral.metric; document.getElementById('pwsTemp').innerText = `${obs.temp} °C`; const feelsLike = obs.temp <= 10 ? obs.windChill : obs.heatIndex; document.getElementById('pwsFeelsLike').innerText = `Gefühlt: ${feelsLike} °C`; document.getElementById('pwsHumidity').innerText = `${obsGeneral.humidity} %`; document.getElementById('pwsDewpt').innerText = `Taupunkt: ${obs.dewpt} °C`; document.getElementById('pwsPressure').innerText = `${obs.pressure} hPa`; const uvVal = obsGeneral.uv ?? 0; const uvInfo = getUvInfo(uvVal); document.getElementById('pwsUv').innerText = uvVal; const badge = document.getElementById('pwsUvBadge'); badge.innerText = uvInfo.text; badge.className = `inline-block text-[10px] px-2.5 py-0.5 rounded font-bold ${uvInfo.color}`; document.getElementById('pwsWind').innerText = `${obs.windSpeed} km/h`; document.getElementById('pwsGust').innerText = `Böen: ${obs.windGust} km/h`; const dir = obsGeneral.winddir; document.getElementById('pwsWindText').innerText = degToCompass(dir); document.getElementById('pwsWindDir').innerText = `Azimut: ${dir ?? '--'} °`; if (dir !== null && dir !== undefined) { document.getElementById('windArrow').style.transform = `rotate(${dir + 135}deg)`; } document.getElementById('pwsPrecip').innerText = `${obs.precipTotal} mm`; document.getElementById('pwsPrecipRate').innerText = `Aktuell: ${obs.precipRate} mm/h`; document.getElementById('pwsSolar').innerText = `${obsGeneral.solarRadiation ?? 0} W/m²`; document.getElementById('pwsObsTime').innerText = obsGeneral.obsTimeLocal || 'Aktuell'; statusMsg.innerText = 'Status: Online • Live-Daten aktiv'; statusDot.className = 'w-2 h-2 rounded-full bg-emerald-400 animate-ping'; } else { statusMsg.innerText = 'Keine Messwerte verfügbar.'; } } catch (error) { console.error('Fehler:', error); statusMsg.innerText = 'Verbindungsfehler beim Abruf.'; statusDot.className = 'w-2 h-2 rounded-full bg-red-500'; } }
